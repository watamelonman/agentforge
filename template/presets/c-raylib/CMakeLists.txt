cmake_minimum_required(VERSION 3.21)
project(@NAME@ C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(FORGE_HEADLESS_ONLY "Build only core/sim/tests (no raylib needed)" OFF)
add_compile_options(-Wall -Wextra -Werror -Wshadow -Wconversion)

# core + sim never link raylib: they are the headless, deterministic, testable layers.
file(GLOB CORE_SRC CONFIGURE_DEPENDS src/core/*.c)
file(GLOB SIM_SRC  CONFIGURE_DEPENDS src/sim/*.c)
add_library(core STATIC ${CORE_SRC})
target_include_directories(core PUBLIC src)
if(SIM_SRC)
  add_library(sim STATIC ${SIM_SRC})
  target_link_libraries(sim PUBLIC core)
endif()

enable_testing()
file(GLOB TEST_SRC CONFIGURE_DEPENDS tests/*.c)
add_executable(test_runner ${TEST_SRC})
target_link_libraries(test_runner PRIVATE core $<$<TARGET_EXISTS:sim>:sim>)
add_test(NAME unit COMMAND test_runner)

if(NOT FORGE_HEADLESS_ONLY)
  find_package(raylib 5.0 REQUIRED)
  file(GLOB GAME_SRC CONFIGURE_DEPENDS src/game/*.c)
  if(GAME_SRC)
    add_library(game MODULE ${GAME_SRC})   # hot-reloadable game.so
    target_link_libraries(game PRIVATE sim raylib)
    set_target_properties(game PROPERTIES C_VISIBILITY_PRESET hidden PREFIX "")
  endif()
  file(GLOB PLATFORM_SRC CONFIGURE_DEPENDS src/platform/*.c)
  if(PLATFORM_SRC)
    add_executable(platform ${PLATFORM_SRC})
    target_link_libraries(platform PRIVATE core raylib ${CMAKE_DL_LIBS})
  endif()
endif()
